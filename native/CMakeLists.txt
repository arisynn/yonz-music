cmake_minimum_required(VERSION 3.22)
project(yono_quickjs C)
set(CMAKE_C_STANDARD 11)
set(QJS_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(quickjs)
add_executable(yono_qjs yono_qjs.c)
target_link_libraries(yono_qjs PRIVATE qjs m)
set_target_properties(yono_qjs PROPERTIES OUTPUT_NAME "qjs")
target_link_options(yono_qjs PRIVATE "-Wl,-z,max-page-size=16384")
